How this deck works

<script type="module" src="…/embed/parsons.js"></script>

<section>
  <parsons-puzzle id="sum" src="puzzles.json" index="0"
                  students-indent check reset theme="light"></parsons-puzzle>
  <span class="fragment" data-parsons-solution="sum"></span>
</section>

deck.on('fragmentshown', (e) => {
  const id = e.fragment.dataset.parsonsSolution;
  if (id) document.getElementById(id).showSolution();
});
// re-centre when a puzzle loads or changes size
const ro = new ResizeObserver(() => deck.layout());
document.querySelectorAll('parsons-puzzle').forEach((p) => ro.observe(p));

Puzzle files use the same format as ParsonXam. See the component docs.
